create table public.posts (
  id uuid not null default gen_random_uuid (),
  user_id uuid null,
  title text not null,
  content text null,
  media_url text null,
  media_type text null,
  category text null default 'General'::text,
  author text null,
  author_avatar text null,
  likes_count integer null default 0,
  comments_count integer null default 0,
  shares_count integer null default 0,
  timestamp timestamp with time zone null default now(),
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint posts_pkey primary key (id),
  constraint posts_user_id_fkey foreign KEY (user_id) references profiles (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_posts_user_id on public.posts using btree (user_id) TABLESPACE pg_default;

create index IF not exists idx_posts_timestamp on public.posts using btree ("timestamp") TABLESPACE pg_default;

create trigger update_posts_updated_at BEFORE
update on posts for EACH row
execute FUNCTION update_updated_at ();
